以 nlme 为参照，提供 gstat 的等价实现，目的是搞明白 gstat 的内部实现原理。为什么这么做？nlme 的方式具有一般性，容易推广到到空间广义线性混合效应模型。

根据 nlme 结果，可以自己写代码实现简单克里金插值预测

多元正态分布的边际分布、条件分布

采用 REML 方法拟合模型，gstat 包的 REML 方法，要求超参数是已知的，具体算法过程见 gstat 包

library(sf)
library(abind)
library(stars)
# 准备数据
rongelap_sf <- st_as_sf(rongelap, coords = c("cX", "cY"), dim = "XY")
rongelap_coastline_sf <- st_as_sf(rongelap_coastline, coords = c("cX", "cY"), dim = "XY")
rongelap_coastline_sfp <- st_cast(st_combine(st_geometry(rongelap_coastline_sf)), "POLYGON")
# 网格化
rongelap_grid_stars <- st_bbox(rongelap_coastline_sfp) |>
  st_as_stars(dx = 32, dy = 16) |>
  st_crop(rongelap_coastline_sfp)

# 变差
library(gstat)
fit_rongelap_gstat <- fit.variogram.reml(log(counts / time) ~ 1,
  locations = ~ cX + cY, data = rongelap,
  model = vgm(psill = 1, model = "Exp", range = 169.7472, nugget = 1)
)

# 预测
fit_krige_pred <- krige(
  formula = log(counts / time) ~ 1, 
  locations = rongelap_sf,
  newdata = rongelap_grid_stars, 
  model = fit_rongelap_gstat
)
